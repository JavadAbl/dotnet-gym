   941→function nextMemberCode(members: Member[]): string {
   942→  const max = members.reduce((acc, m) => {
   943→    const suffix = m.code.startsWith("M-") ? Number.parseInt(m.code.slice(2), 10) : Number.NaN;
   944→    return Number.isFinite(suffix) && suffix > acc ? suffix : acc;
   945→  }, 0);
   946→  return `M-${String(max + 1).padStart(5, "0")}`;
   947→}

   949→function duplicateFieldErrors(db: DemoDb, phone: string | null | undefined, card: string | null | undefined, excludeId?: string) {
   950→  const errors: Record<string, string[]> = {};
   951→  if (phone && db.members.some((m) => m.phoneNumber === phone && m.id !== excludeId)) {
   952→    errors.phoneNumber = [`Phone number ${phone} is already registered to another member.`];
   953→  }
   954→  if (card && db.members.some((m) => m.cardCode === card && m.id !== excludeId)) {
   955→    errors.cardCode = [`Card ${card} is already assigned to another member.`];
   956→  }
   957→  return errors;
   958→}

   960→export async function demoSearchMembers(query: MemberQuery): Promise<PagedResult<Member>> {
   961→  await wait(180);
   962→  const db = loadDb();
   963→  const page = query.page && query.page > 0 ? query.page : 1;
   964→  const pageSize = query.pageSize && query.pageSize > 0 ? Math.min(query.pageSize, 100) : 20;
   965→
   966→  const term = query.search?.trim().toLowerCase() ?? "";
   967→  let items = [...db.members];
   968→
   969→  if (term) {
   970→    items = items.filter(
   971→      (m) =>
   972→        m.fullName.toLowerCase().includes(term) ||
   973→        m.code.toLowerCase().includes(term) ||
   974→        (m.phoneNumber ?? "").includes(term) ||
   975→        (m.cardCode ?? "").toLowerCase().includes(term),
   976→    );
   977→  }
   978→  if (query.status) {
   979→    items = items.filter((m) => m.status === query.status);
   980→  }
   981→  if (query.branchId) {
   982→    items = items.filter((m) => m.homeBranchId === query.branchId);
   983→  }
   984→
   985→  items.sort((a, b) => (a.joinDateUtc < b.joinDateUtc ? 1 : -1));
   986→  const totalCount = items.length;
   987→  const paged = items.slice((page - 1) * pageSize, page * pageSize);
   988→  return { items: paged, page, pageSize, totalCount };
   989→}
