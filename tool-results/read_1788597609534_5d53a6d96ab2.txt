   890→function seedDemoBuffet(db: DemoDb): void {
   891→  const north = db.branches.find((b) => b.nameEn === "North Branch")!.id;
   892→  const south = db.branches.find((b) => b.nameEn === "South Branch")!.id;
   893→  const memberByCode = (code: string) => db.members.find((m) => m.code === code)!;
   894→  const productByName = (prefix: string) => db.products.find((p) => p.nameEn.startsWith(prefix))!;
   895→
   896→  const water = productByName("Water");
   897→  const proteinBar = productByName("Protein");
   898→  const energy = productByName("Energy");
   899→  const priceOf = (product: BuffetProduct, branchId: string) =>
   900→    product.prices.find((p) => p.branchId === branchId)!.priceToman;
   901→
   902→  // Wallet debit with a linked BuffetPurchase receipt — mirrors the endpoint orchestration.
   903→  const placeOrder = (
   904→    member: Member,
   905→    branchId: string,
   906→    paymentMethod: PaymentMethod,
   907→    lines: Array<{ product: BuffetProduct; quantity: number }>,
   908→    at: string,
   909→    note: string,
   910→  ): void => {
   911→    const branch = db.branches.find((b) => b.id === branchId)!;
   912→    const items = lines.map(({ product, quantity }) => {
   913→      const unitPriceToman = priceOf(product, branchId);
   914→      return {
   915→        productId: product.id,
   916→        nameEn: product.nameEn,
   917→        nameFa: product.nameFa,
   918→        quantity,
   919→        unitPriceToman,
   920→        lineTotalToman: unitPriceToman * quantity,
   921→      };
   922→    });
   923→    const totalToman = items.reduce((sum, item) => sum + item.lineTotalToman, 0);
   924→
   925→    const receiptId = uid();
   926→    db.receipts.push({
   927→      id: receiptId,
   928→      memberId: member.id,
   929→      branchId,
   930→      memberPackageId: null, // buffet orders are not tied to any exercise package.
   931→      kind: "BuffetPurchase",
   932→      paymentMethod,
   933→      amountToman: totalToman,
   934→      note,
   935→      createdAtUtc: at,
   936→      memberCode: member.code,
   937→      memberFullName: member.fullName,
   938→      branchNameEn: branch.nameEn,
   939→      branchNameFa: branch.nameFa,
   940→      packageNameEn: "—",
   941→      packageNameFa: "—",
   942→    });
   943→
   944→    if (paymentMethod === "Wallet") {
   945→      const account = db.walletAccounts.find((a) => a.memberId === member.id);
   946→      if (!account) throw new Error(`Seed error: no wallet account for ${member.code}`);
   947→      if (account.balanceToman < totalToman) throw new Error("Seed error: insufficient wallet balance");
   948→      account.balanceToman -= totalToman;
   949→      db.walletTransactions.push({
   950→        id: uid(),
   951→        walletAccountId: account.id,
   952→        amountToman: -totalToman,
   953→        kind: "BuffetPurchase",
   954→        balanceAfter: account.balanceToman,
   955→        relatedReceiptId: receiptId,
   956→        note,
   957→        createdAtUtc: at,
   958→      });
   959→    }
   960→
   961→    db.buffetOrders.push({
   962→      id: uid(),
   963→      memberId: member.id,
   964→      branchId,
   965→      paymentMethod,
   966→      totalToman,
   967→      receiptId,
   968→      note,
   969→      createdAtUtc: at,
   970→      items,
   971→    });
   972→  };
   973→
   974→  // 1) Ali — wallet @ North, yesterday: 2× water + 1× protein bar = 115,000 T.
   975→  placeOrder(memberByCode("M-00001"), north, "Wallet", [{ product: water, quantity: 2 }, { product: proteinBar, quantity: 1 }],
   976→    daysFromNow(-1), "Buffet order (seed).");
   977→
   978→  // 2) Sara — cash @ South, ~90 minutes ago: 1× energy drink + 1× protein bar = 155,000 T.
   979→  placeOrder(memberByCode("M-00002"), south, "Cash", [{ product: energy, quantity: 1 }, { product: proteinBar, quantity: 1 }],
   980→    minutesFromNow(-90), "Buffet order (seed).");
   981→
   982→  // 3) Maryam — wallet @ North, 2 hours ago: 1× water = 15,000 T.
   983→  placeOrder(memberByCode("M-00004"), north, "Wallet", [{ product: water, quantity: 1 }],
   984→    minutesFromNow(-120), "Buffet order (seed).");
   985→}
