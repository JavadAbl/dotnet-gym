   677→function nextMemberCode(members: Member[]): string {
   678→  const max = members.reduce((acc, m) => {
   679→    const suffix = m.code.startsWith("M-") ? Number.parseInt(m.code.slice(2), 10) : Number.NaN;
   680→    return Number.isFinite(suffix) && suffix > acc ? suffix : acc;
   681→  }, 0);
   682→  return `M-${String(max + 1).padStart(5, "0")}`;
   683→}

   685→function duplicateFieldErrors(db: DemoDb, phone: string | null | undefined, card: string | null | undefined, excludeId?: string) {
   686→  const errors: Record<string, string[]> = {};
   687→  if (phone && db.members.some((m) => m.phoneNumber === phone && m.id !== excludeId)) {
   688→    errors.phoneNumber = [`Phone number ${phone} is already registered to another member.`];
   689→  }
   690→  if (card && db.members.some((m) => m.cardCode === card && m.id !== excludeId)) {
   691→    errors.cardCode = [`Card ${card} is already assigned to another member.`];
   692→  }
   693→  return errors;
   694→}

   696→export async function demoSearchMembers(query: MemberQuery): Promise<PagedResult<Member>> {
   697→  await wait(180);
   698→  const db = loadDb();
   699→  const page = query.page && query.page > 0 ? query.page : 1;
   700→  const pageSize = query.pageSize && query.pageSize > 0 ? Math.min(query.pageSize, 100) : 20;
   701→
   702→  const term = query.search?.trim().toLowerCase() ?? "";
   703→  let items = [...db.members];
   704→
   705→  if (term) {
   706→    items = items.filter(
   707→      (m) =>
   708→        m.fullName.toLowerCase().includes(term) ||
   709→        m.code.toLowerCase().includes(term) ||
   710→        (m.phoneNumber ?? "").includes(term) ||
   711→        (m.cardCode ?? "").toLowerCase().includes(term),
   712→    );
   713→  }
   714→  if (query.status) {
   715→    items = items.filter((m) => m.status === query.status);
   716→  }
   717→  if (query.branchId) {
   718→    items = items.filter((m) => m.homeBranchId === query.branchId);
   719→  }
   720→
   721→  items.sort((a, b) => (a.joinDateUtc < b.joinDateUtc ? 1 : -1));
   722→  const totalCount = items.length;
   723→  const paged = items.slice((page - 1) * pageSize, page * pageSize);
   724→  return { items: paged, page, pageSize, totalCount };
   725→}

   734→export async function demoCreateMember(model: MemberWriteModel): Promise<Member> {
   735→  await wait(260);
   736→  const db = loadDb();
   737→
   738→  const errors: Record<string, string[]> = {};
   739→  if (!model.fullName.trim()) errors.fullName = ["Full name is required."];
   740→  if (!model.homeBranchId || !db.branches.some((b) => b.id === model.homeBranchId && b.isActive)) {
   741→    errors.homeBranchId = ["Unknown or inactive branch."];
   742→  }
   743→  Object.assign(errors, duplicateFieldErrors(db, model.phoneNumber?.trim() || null, model.cardCode?.trim() || null));
   744→  if (Object.keys(errors).length > 0) {
   745→    throw new GmsApiError(400, "validation.failed", "One or more validation errors occurred.", errors);
   746→  }
   747→
   748→  const member: Member = {
   749→    id: uid(),
   750→    code: nextMemberCode(db.members),
   751→    fullName: model.fullName.trim(),
   752→    phoneNumber: model.phoneNumber?.trim() || null,
   753→    gender: model.gender ?? "Unknown",
   754→    birthDateUtc: model.birthDate ? new Date(model.birthDate).toISOString() : null,
   755→    joinDateUtc: new Date().toISOString(),
   756→    homeBranchId: model.homeBranchId,
   757→    cardCode: model.cardCode?.trim() || null,
   758→    status: "Active",
   759→    notes: model.notes?.trim() || null,
   760→  };
   761→  db.members.push(member);
   762→  saveDb(db);
   763→  return member;
   764→}

   767→  await wait(260);
   768→  const db = loadDb();
   769→  const member = db.members.find((m) => m.id === id);
   770→  if (!member) throw new GmsApiError(404, "member.not_found", "Member not found.");

   772→  const errors: Record<string, string[]> = {};
   773→  if (!model.fullName.trim()) errors.fullName = ["Full name is required."];
   774→  if (!model.homeBranchId || !db.branches.some((b) => b.id === model.homeBranchId && b.isActive)) {
   775→    errors.homeBranchId = ["Unknown or inactive branch."];

   777→  Object.assign(
   778→    errors,
   779→    duplicateFieldErrors(db, model.phoneNumber?.trim() || null, model.cardCode?.trim() || null, id),
   780→  );
   781→  if (Object.keys(errors).length > 0) {
