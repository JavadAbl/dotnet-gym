     1→"use client";
     2→
     3→import * as React from "react";
     4→import {
     5→  Banknote,
     6→  ChevronLeft,
     7→  ChevronRight,
     8→  CircleAlert,
     9→  CircleCheck,
    10→  CupSoda,

    18→import { toast } from "sonner";

    21→import { Card, CardContent } from "@/components/ui/card";
    22→import { Input } from "@/components/ui/input";
    23→import { Label } from "@/components/ui/label";
    24→import { Skeleton } from "@/components/ui/skeleton";
    25→import {
    26→  Select,
    27→  SelectContent,
    28→  SelectItem,
    29→  SelectTrigger,
    30→  SelectValue,
    31→} from "@/components/ui/select";
    32→import { Tabs, TabsList, TabsTrigger } from "@/components/ui/tabs";
    33→import {
    34→  Table,
    35→  TableBody,
    36→  TableCell,
    37→  TableHead,
    38→  TableHeader,
    39→  TableRow,
    40→} from "@/components/ui/table";
    41→import {
    42→  createBuffetOrder,
    43→  getBuffetSummary,
    44→  getWalletSnapshot,
    45→  listProducts,
    46→  searchBuffetOrders,
    47→  searchMembers,
    48→} from "@/lib/gms/api";
    49→import type {
    50→  Branch,
    51→  BuffetOrder,
    52→  BuffetProduct,
    53→  BuffetSummary,
    54→  Member,
    55→  PaymentMethod,
    56→  WalletSnapshot,
    57→} from "@/lib/gms/types";
    58→import { formatDateTime, formatNumber, formatToman } from "@/lib/gms/jalali";
    59→import { StatusBadge } from "./status-badge";
    60→import { describeError } from "./sell-package-dialog";
    61→import { ProductDialog } from "./product-dialog";
    62→import { useAuth, useI18n } from "./providers";
    63→
    64→interface BuffetViewProps {
    65→  branches: Branch[];
    66→  refreshSignal: number;
    67→}
    68→
    69→const PAGE_SIZE = 10;
    70→const MAX_QUANTITY = 99;
    71→
    72→export function BuffetView({ branches, refreshSignal }: BuffetViewProps) {
    73→  const { t, locale } = useI18n();
    74→  const { session } = useAuth();
    75→  const user = session!.user;
    76→  const isAdmin = user.role === "admin";
    77→  const pinnedBranchId = user.branchId ?? null;
    78→
    79→  const [tab, setTab] = React.useState<"pos" | "orders" | "products">("pos");
    80→  const [refreshBump, setRefreshBump] = React.useState(0);
    81→
    82→  // branch: reception/buffet pinned to their claim, admin picks (check-in pattern)
    83→  const [branchId, setBranchId] = React.useState(pinnedBranchId ?? "");
    84→  const effectiveBranchId = pinnedBranchId ?? branchId;
    85→
    86→  // ---------------------------------------------------------------- KPI summary + catalog
    87→  const [summary, setSummary] = React.useState<BuffetSummary | null>(null);
    88→  const [summaryError, setSummaryError] = React.useState(false);
    89→  const [products, setProducts] = React.useState<BuffetProduct[] | null>(null);
    90→  const [productsError, setProductsError] = React.useState(false);
    91→
    92→  React.useEffect(() => {
    93→    let cancelled = false;
    94→    getBuffetSummary(effectiveBranchId || undefined)
    95→      .then((result) => {
    96→        if (!cancelled) {
    97→          setSummary(result);
    98→          setSummaryError(false);
    99→        }
   100→      })
   101→      .catch(() => {
   102→        if (!cancelled) {
   103→          setSummary(null);
   104→          setSummaryError(true);
   105→        }
   106→      });
   107→    return () => {
   108→      cancelled = true;
   109→    };
   110→  }, [effectiveBranchId, refreshBump, refreshSignal]);
   111→
   112→  React.useEffect(() => {
   113→    let cancelled = false;
   114→    listProducts()
   115→      .then((items) => {
   116→        if (!cancelled) {
   117→          setProducts(items);
   118→          setProductsError(false);
   119→        }
   120→      })
   121→      .catch(() => {
   122→        if (!cancelled) {
   123→          setProducts(null);
   124→          setProductsError(true);
   125→        }
   126→      });
   127→    return () => {
   128→      cancelled = true;
   129→    };
   130→  }, [refreshBump, refreshSignal]);
   131→
   132→  const localizedProductName = React.useCallback(
   133→    (product: { nameEn: string; nameFa: string }) => (locale === "fa" ? product.nameFa : product.nameEn),
   134→    [locale],
   135→  );
   136→
   137→  const priceInBranch = React.useCallback(
   138→    (product: BuffetProduct, branch: string | null | undefined) =>
   139→      branch ? product.prices.find((p) => p.branchId === branch) ?? null : null,
   140→    [],
   141→  );
   142→
   143→  // ---------------------------------------------------------------- POS tab
   144→  const [memberQuery, setMemberQuery] = React.useState("");
   145→  const [memberResults, setMemberResults] = React.useState<Member[]>([]);
   146→  const [memberSearchOpen, setMemberSearchOpen] = React.useState(false);
   147→  const [selectedMember, setSelectedMember] = React.useState<Member | null>(null);
   148→  const [quantities, setQuantities] = React.useState<Record<string, number>>({});
   149→  const [paymentMethod, setPaymentMethod] = React.useState<PaymentMethod>("Cash");
   150→  const [wallet, setWallet] = React.useState<WalletSnapshot | null>(null);
   151→  const [note, setNote] = React.useState("");
   152→  const [submitting, setSubmitting] = React.useState(false);
   153→  const [orderError, setOrderError] = React.useState<string | null>(null);
   154→  const [lastOrder, setLastOrder] = React.useState<BuffetOrder | null>(null);
   155→
   156→  React.useEffect(() => {
   157→    // switching the branch changes prices — reset the cart
   158→    setQuantities({});
   159→    setLastOrder(null);
   160→    setOrderError(null);
   161→  }, [effectiveBranchId]);
   162→
   163→  React.useEffect(() => {
   164→    if (!selectedMember || paymentMethod !== "Wallet") {
   165→      setWallet(null);
   166→      return;
   167→    }
   168→    let cancelled = false;
   169→    getWalletSnapshot(selectedMember.id)
   170→      .then((snapshot) => {
   171→        if (!cancelled) setWallet(snapshot);
   172→      })
   173→      .catch(() => {
   174→        if (!cancelled) setWallet(null);
   175→      });
   176→    return () => {
   177→      cancelled = true;
   178→    };
   179→  }, [selectedMember, paymentMethod, refreshBump]);
   180→
   181→  // debounced member search
   182→  React.useEffect(() => {
   183→    const handle = setTimeout(() => {
   184→      const term = memberQuery.trim();
   185→      if (term.length === 0) {
   186→        setMemberResults([]);
   187→        setMemberSearchOpen(false);
   188→        return;
   189→      }
   190→      searchMembers({ search: term, page: 1, pageSize: 6 })
   191→        .then((res) => {
   192→          setMemberResults(res.items);
   193→          setMemberSearchOpen(true);
   194→        })
   195→        .catch(() => setMemberResults([]));
   196→    }, 250);
   197→    return () => clearTimeout(handle);
   198→  }, [memberQuery]);
   199→
   200→  const cartLines = React.useMemo(() => {
   201→    if (!products) return [];
   202→    return Object.entries(quantities)
   203→      .filter(([, quantity]) => quantity > 0)
   204→      .map(([productId, quantity]) => {
   205→        const product = products.find((p) => p.id === productId)!;
   206→        const price = priceInBranch(product, effectiveBranchId);
   207→        const unitPriceToman = price?.priceToman ?? 0;
   208→        return { product, quantity, unitPriceToman, lineTotalToman: unitPriceToman * quantity };
   209→      });
   210→  }, [quantities, products, effectiveBranchId, priceInBranch]);
   211→
   212→  const totalToman = cartLines.reduce((sum, line) => sum + line.lineTotalToman, 0);
   213→  const walletBalance = wallet?.balanceToman ?? 0;
   214→  const walletInsufficient = paymentMethod === "Wallet" && selectedMember !== null && wallet !== null && walletBalance < totalToman;
   215→  const memberInactive = selectedMember !== null && selectedMember.status !== "Active";
   216→  const canSubmit =
   217→    selectedMember !== null &&
   218→    Boolean(effectiveBranchId) &&
   219→    cartLines.length > 0 &&
   220→    totalToman > 0 &&
   221→    !memberInactive &&
   222→    !walletInsufficient &&
   223→    !submitting;
   224→
   225→  function setQuantity(productId: string, quantity: number) {
   226→    setQuantities((prev) => {
   227→      const next = { ...prev };
   228→      if (quantity <= 0) {
   229→        delete next[productId];
   230→      } else {
   231→        next[productId] = Math.min(quantity, MAX_QUANTITY);
   232→      }
   233→      return next;
   234→    });
   235→    setLastOrder(null);
   236→    setOrderError(null);
   237→  }
   238→
   239→  function pickMember(member: Member) {
   240→    setSelectedMember(member);
   241→    setMemberSearchOpen(false);
   242→    setMemberQuery("");
   243→  }
   244→
   245→  async function handleSubmitOrder(event: React.FormEvent) {
   246→    event.preventDefault();
   247→    if (!selectedMember || !effectiveBranchId || cartLines.length === 0) return;
   248→
   249→    setSubmitting(true);
   250→    setOrderError(null);
   251→    try {
   252→      const order = await createBuffetOrder({
   253→        memberId: selectedMember.id,
   254→        branchId: effectiveBranchId,
   255→        paymentMethod,
   256→        items: cartLines.map((line) => ({ productId: line.product.id, quantity: line.quantity })),
   257→        note: note.trim() || undefined,
   258→      });
   259→      setLastOrder(order);
   260→      toast.success(t("buffet.pos.success.toast", { amount: formatToman(order.totalToman, locale) }));
   261→      setQuantities({});
   262→      setNote("");
   263→      setRefreshBump((n) => n + 1);
   264→    } catch (err) {
   265→      setOrderError(describeError(t, err));
   266→    } finally {
   267→      setSubmitting(false);
   268→    }
   269→  }
   270→
   271→  // ---------------------------------------------------------------- orders tab
   272→  const [orderSearchInput, setOrderSearchInput] = React.useState("");
   273→  const [orderSearch, setOrderSearch] = React.useState("");
   274→  const [orderBranchFilter, setOrderBranchFilter] = React.useState<string>("all");
   275→  const [orderMethodFilter, setOrderMethodFilter] = React.useState<string>("all");
   276→  const [orderPage, setOrderPage] = React.useState(1);
   277→  const [orders, setOrders] = React.useState<BuffetOrder[] | null>(null);
   278→  const [orderCount, setOrderCount] = React.useState(0);
   279→  const [ordersLoading, setOrdersLoading] = React.useState(false);
   280→  const [ordersError, setOrdersError] = React.useState(false);
   281→
   282→  React.useEffect(() => {
   283→    const handle = setTimeout(() => {
   284→      setOrderSearch(orderSearchInput.trim());
   285→      setOrderPage(1);
   286→    }, 300);
   287→    return () => clearTimeout(handle);
   288→  }, [orderSearchInput]);
   289→
   290→  React.useEffect(() => {
   291→    if (tab !== "orders") return;
   292→    let cancelled = false;
   293→    setOrdersLoading(true);
   294→    searchBuffetOrders({
   295→      search: orderSearch || undefined,
   296→      branchId: (isAdmin ? (orderBranchFilter !== "all" ? orderBranchFilter : undefined) : effectiveBranchId || undefined),
   297→      paymentMethod: orderMethodFilter !== "all" ? (orderMethodFilter as PaymentMethod) : undefined,
   298→      page: orderPage,
   299→      pageSize: PAGE_SIZE,
   300→    })
   301→      .then((result) => {
   302→        if (cancelled) return;
   303→        setOrders(result.items);
   304→        setOrderCount(result.totalCount);
   305→        setOrdersError(false);
   306→      })
   307→      .catch(() => {
   308→        if (!cancelled) {
   309→          setOrders([]);
   310→          setOrderCount(0);
   311→          setOrdersError(true);
   312→        }
   313→      })
   314→      .finally(() => {
   315→        if (!cancelled) setOrdersLoading(false);
   316→      });
   317→    return () => {
   318→      cancelled = true;
   319→    };
   320→  }, [tab, orderSearch, orderBranchFilter, orderMethodFilter, orderPage, refreshBump, refreshSignal, isAdmin, effectiveBranchId]);
   321→
   322→  const orderTotalPages = Math.max(1, Math.ceil(orderCount / PAGE_SIZE));
   323→
   324→  // ---------------------------------------------------------------- products tab (admin)
   325→  const [productDialogOpen, setProductDialogOpen] = React.useState(false);
   326→  const [editingProduct, setEditingProduct] = React.useState<BuffetProduct | null>(null);
   327→
   328→  const activeBranches = branches.filter((b) => b.isActive);
   329→
   330→  return (
   331→    <div className="space-y-6">
   332→      <div>
   333→        <h2 className="text-2xl font-bold tracking-tight">{t("buffet.title")}</h2>
   334→        <p className="text-sm text-muted-foreground">{t("buffet.subtitle")}</p>
   335→      </div>
   336→
   337→      {/* KPI chips */}
   338→      <div className="grid grid-cols-2 gap-3 xl:grid-cols-4">
   339→        {[
   340→          {
   341→            key: "buffet.kpi.ordersToday",
   342→            icon: ShoppingBasket,
   343→            value: summary ? formatNumber(summary.ordersToday, locale) : null,
   344→            tone: "bg-emerald-600",
   345→          },
   346→          {
   347→            key: "buffet.kpi.revenueToday",
   348→            icon: ReceiptText,
   349→            value: summary ? formatToman(summary.revenueTodayToman, locale) : null,
   350→            tone: "bg-teal-600",
   351→          },
   352→          {
   353→            key: "buffet.kpi.cashToday",
   354→            icon: Banknote,
   355→            value: summary ? formatToman(summary.cashRevenueTodayToman, locale) : null,
   356→            tone: "bg-amber-600",
   357→          },
   358→          {
   359→            key: "buffet.kpi.walletToday",
   360→            icon: WalletIcon,
   361→            value: summary ? formatToman(summary.walletRevenueTodayToman, locale) : null,
   362→            tone: "bg-violet-600",
   363→          },
   364→        ].map((chip) => {
   365→          const Icon = chip.icon;
   366→          return (
   367→            <Card key={chip.key} className="py-0">
   368→              <CardContent className="flex items-center gap-3 p-4">
   369→                <span className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-xl ${chip.tone} text-white`}>
   370→                  <Icon className="h-4.5 w-4.5" />
   371→                </span>
   372→                <div className="min-w-0">
   373→                  {chip.value === null && !summaryError ? (
   374→                    <Skeleton className="mb-1 h-6 w-20" />
   375→                  ) : (
   376→                    <p className="truncate text-lg font-bold tabular-nums">{chip.value ?? "—"}</p>
   377→                  )}
   378→                  <p className="truncate text-xs text-muted-foreground">{t(chip.key)}</p>
   379→                </div>
   380→              </CardContent>
   381→            </Card>
   382→          );
   383→        })}
   384→      </div>
   385→
   386→      <Tabs value={tab} onValueChange={(value) => setTab(value as typeof tab)}>
   387→        <TabsList className="w-full sm:w-auto">
   388→          <TabsTrigger value="pos" className="gap-1.5">
   389→            <CupSoda className="h-4 w-4" />
   390→            {t("buffet.tab.pos")}
   391→          </TabsTrigger>
   392→          <TabsTrigger value="orders" className="gap-1.5">
   393→            <ReceiptText className="h-4 w-4" />
   394→            {t("buffet.tab.orders")}
   395→          </TabsTrigger>
   396→          {isAdmin ? (
   397→            <TabsTrigger value="products" className="gap-1.5">
   398→              <ShoppingBasket className="h-4 w-4" />
   399→              {t("buffet.tab.products")}
   400→            </TabsTrigger>
   401→          ) : null}
   402→        </TabsList>
   403→      </Tabs>
   404→
   405→      {/* ---------------------------------------------------------------- POS tab */}
   406→      {tab === "pos" ? (
   407→        <div className="grid grid-cols-1 gap-4 lg:grid-cols-5">
   408→          {/* member + products */}
   409→          <div className="space-y-4 lg:col-span-3">
   410→            <div className="space-y-1.5">
   411→              <Label>{t("buffet.pos.member")}</Label>
   412→              {selectedMember ? (
   413→                <div className="flex items-center justify-between gap-2 rounded-lg border bg-card px-3 py-2">
   414→                  <div className="flex min-w-0 items-center gap-2">
   415→                    <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-emerald-100 text-xs font-bold text-emerald-800 dark:bg-emerald-900 dark:text-emerald-200">
   416→                      {selectedMember.fullName.slice(0, 1)}
   417→                    </span>
   418→                    <div className="min-w-0">
   419→                      <p className="truncate text-sm font-medium">{selectedMember.fullName}</p>
   420→                      <p className="font-mono text-xs text-muted-foreground" dir="ltr">
   421→                        {selectedMember.code}
   422→                      </p>
   423→                    </div>
   424→                  </div>
   425→                  <div className="flex items-center gap-2">
   426→                    <StatusBadge status={selectedMember.status} />
   427→                    <Button type="button" variant="ghost" size="sm" onClick={() => setSelectedMember(null)}>
   428→                      ×
   429→                    </Button>
   430→                  </div>
   431→                </div>
   432→              ) : (
   433→                <div className="relative">
   434→                  <Search className="absolute top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground start-3" />
   435→                  <Input
   436→                    value={memberQuery}
   437→                    onChange={(e) => setMemberQuery(e.target.value)}
   438→                    placeholder={t("buffet.pos.member.placeholder")}
   439→                    className="ps-9"
   440→                  />
   441→                  {memberSearchOpen ? (
   442→                    <div className="absolute inset-x-0 top-full z-30 mt-1 overflow-hidden rounded-lg border bg-popover shadow-md">
   443→                      {memberResults.length === 0 ? (
   444→                        <p className="px-3 py-2 text-sm text-muted-foreground">{t("buffet.pos.member.none")}</p>
   445→                      ) : (
   446→                        memberResults.map((member) => (
   447→                          <button
   448→                            key={member.id}
   449→                            type="button"
   450→                            onClick={() => pickMember(member)}
   451→                            className="flex w-full items-center justify-between gap-2 px-3 py-2 text-start hover:bg-accent"
   452→                          >
   453→                            <span className="min-w-0">
   454→                              <span className="block truncate text-sm font-medium">{member.fullName}</span>
   455→                              <span className="block font-mono text-xs text-muted-foreground" dir="ltr">
   456→                                {member.code}
   457→                              </span>
   458→                            </span>
   459→                            <StatusBadge status={member.status} />
   460→                          </button>
   461→                        ))
   462→                      )}
   463→                    </div>
   464→                  ) : null}
   465→                </div>
   466→              )}
   467→              {memberInactive ? (
   468→                <p className="flex items-center gap-1.5 text-xs text-destructive">
   469→                  <CircleAlert className="h-3.5 w-3.5" />
   470→                  {t("buffet.pos.activeMemberOnly")}
   471→                </p>
   472→              ) : null}
   473→            </div>
   474→
   475→            {/* branch: pinned for reception/buffet, admin picks (affects per-branch prices) */}
   476→            <div className="space-y-1.5">
   477→              <Label>{t("buffet.pos.branch")}</Label>
   478→              {pinnedBranchId ? (
   479→                <div className="flex h-9 items-center rounded-md border bg-muted/40 px-3 text-sm text-muted-foreground">
   480→                  {(() => {
   481→                    const branch = branches.find((b) => b.id === pinnedBranchId);
   482→                    return branch ? (locale === "fa" ? branch.nameFa : branch.nameEn) : t("checkin.branch.pinned");
   483→                  })()}
   484→                </div>
   485→              ) : (
   486→                <Select
   487→                  value={branchId || undefined}
   488→                  onValueChange={(value) => {
   489→                    setBranchId(value);
   490→                  }}
   491→                >
   492→                  <SelectTrigger className="sm:w-60">
   493→                    <SelectValue placeholder={t("buffet.pos.branch.placeholder")} />
   494→                  </SelectTrigger>
   495→                  <SelectContent>
   496→                    {activeBranches.map((branch) => (
   497→                      <SelectItem key={branch.id} value={branch.id}>
   498→                        {locale === "fa" ? branch.nameFa : branch.nameEn}
   499→                      </SelectItem>
   500→                    ))}
   501→                  </SelectContent>
   502→                </Select>
   503→              )}
   504→            </div>
   505→
   506→            {/* product grid */}
   507→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   508→              {products === null && !productsError
   509→                ? Array.from({ length: 4 }).map((_, index) => <Skeleton key={index} className="h-24 w-full" />)
   510→                : null}
   511→              {productsError ? (
   512→                <p className="text-sm text-destructive sm:col-span-2">{t("common.error.load")}</p>
   513→              ) : null}
   514→              {products
   515→                ?.filter((product) => product.isActive)
   516→                .map((product) => {
   517→                  const price = priceInBranch(product, effectiveBranchId);
   518→                  const quantity = quantities[product.id] ?? 0;
   519→                  return (
   520→                    <Card key={product.id} className={`py-0 ${price ? "" : "opacity-60"}`}>
   521→                      <CardContent className="flex items-center justify-between gap-3 p-4">
   522→                        <div className="min-w-0">
   523→                          <p className="truncate text-sm font-semibold">{localizedProductName(product)}</p>
   524→                          <p className="mt-0.5 text-sm font-medium tabular-nums text-emerald-700 dark:text-emerald-400">
   525→                            {price ? formatToman(price.priceToman, locale) : t("buffet.pos.notPriced")}
   526→                          </p>
   527→                        </div>
   528→                        {price ? (
   529→                          <div className="flex shrink-0 items-center gap-1.5">
   530→                            <Button
   531→                              type="button"
   532→                              variant="outline"
   533→                              size="icon"
   534→                              className="h-8 w-8"
   535→                              aria-label={t("buffet.pos.decrease")}
   536→                              disabled={quantity === 0}
   537→                              onClick={() => setQuantity(product.id, quantity - 1)}
   538→                            >
   539→                              <Minus className="h-3.5 w-3.5" />
   540→                            </Button>
   541→                            <span className="w-8 text-center text-sm font-bold tabular-nums" dir="ltr">
   542→                              {formatNumber(quantity, locale)}
   543→                            </span>
   544→                            <Button
   545→                              type="button"
   546→                              variant="outline"
   547→                              size="icon"
   548→                              className="h-8 w-8"
   549→                              aria-label={t("buffet.pos.increase")}
   550→                              disabled={!effectiveBranchId || quantity >= MAX_QUANTITY}
   551→                              onClick={() => setQuantity(product.id, quantity + 1)}
   552→                            >
