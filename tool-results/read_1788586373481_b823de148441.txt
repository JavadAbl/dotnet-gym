   583→function seedDemoAttendance(db: DemoDb): void {
   584→  const north = db.branches.find((b) => b.nameEn === "North Branch")!.id;
   585→  const south = db.branches.find((b) => b.nameEn === "South Branch")!.id;
   586→  const memberByCard = (card: string) => db.members.find((m) => m.cardCode === card)!;
   587→
   588→  const ali = memberByCard("RF-0001");
   589→  const sara = memberByCard("RF-0002");
   590→  const maryam = memberByCard("RF-0004");
   591→
   592→  // Active member packages (mirrors the backend resolving through the Sales contract).
   593→  const usable = (memberId: string, branchId: string) =>
   594→    db.memberPackages.find(
   595→      (mp) =>
   596→        mp.memberId === memberId &&
   597→        mp.status === "Active" &&
   598→        mp.expiresAtUtc > new Date().toISOString() &&
   599→        mp.sessionsUsed < mp.sessionsTotal &&
   600→        mp.branchId === branchId,
   601→    );
   602→
   603→  const maryamVip = usable(maryam.id, north);
   604→  const saraBasic = usable(sara.id, south);
   605→  const aliVip = usable(ali.id, north);
   606→  const aliBasicSouth = db.memberPackages.find((mp) => mp.memberId === ali.id && mp.branchId === south);
   607→
   608→  const toEntry = (
   609→    member: Member,
   610→    mp: DemoMemberPackageRecord,
   611→    branchId: string,
   612→    branchNameEn: string,
   613→    branchNameFa: string,
   614→    checkInAtUtc: string,
   615→    checkOutAtUtc: string | null,
   616→  ): DemoAttendanceRecord => {
   617→    const endMs = checkOutAtUtc ? new Date(checkOutAtUtc).getTime() : Date.now();
   618→    return {
   619→      id: uid(),
   620→      memberId: member.id,
   621→      memberPackageId: mp.id,
   622→      branchId,
   623→      checkInAtUtc,
   624→      checkOutAtUtc,
   625→      durationMinutes: Math.max(0, Math.floor((endMs - new Date(checkInAtUtc).getTime()) / 60_000)),
   626→      isPresent: checkOutAtUtc === null,
   627→      memberCode: member.code,
   628→      memberFullName: member.fullName,
   629→      packageNameEn: mp.packageNameEn,
   630→      packageNameFa: mp.packageNameFa,
   631→      branchNameEn,
   632→      branchNameFa,
   633→      sessionsTotal: mp.sessionsTotal,
   634→      sessionsUsed: mp.sessionsUsed,
   635→      sessionsRemaining: Math.max(0, mp.sessionsTotal - mp.sessionsUsed),
   636→    };
   637→  };
   638→
   639→  // 1) Maryam — IN THE GYM NOW @ North (open entry, 35 min)
   640→  if (maryamVip) {
   641→    db.attendance.push(toEntry(maryam, maryamVip, north, "North Branch", "شعبه شمال", minutesAgo(35), null));
   642→  }
   643→
   644→  // 2) Sara — IN THE GYM NOW @ South (open entry, 12 min)
   645→  if (saraBasic) {
   646→    db.attendance.push(toEntry(sara, saraBasic, south, "South Branch", "شعبه جنوب", minutesAgo(12), null));
   647→  }
   648→
   649→  // 3) Ali — visited yesterday @ North (closed)
   650→  if (aliVip) {
   651→    db.attendance.push(
   652→      toEntry(ali, aliVip, north, "North Branch", "شعبه شمال", minutesAgo(26 * 60), minutesAgo(24 * 60)),
   653→    );
   654→  }
   655→
   656→  // 4) Ali — visited 3 days ago @ South on Basic (closed)
   657→  if (aliBasicSouth) {
   658→    db.attendance.push(
   659→      toEntry(ali, aliBasicSouth, south, "South Branch", "شعبه جنوب", minutesAgo(74 * 60), minutesAgo(72 * 60)),

   661→  }

   743→function nextMemberCode(members: Member[]): string {
   744→  const max = members.reduce((acc, m) => {
   745→    const suffix = m.code.startsWith("M-") ? Number.parseInt(m.code.slice(2), 10) : Number.NaN;
   746→    return Number.isFinite(suffix) && suffix > acc ? suffix : acc;
   747→  }, 0);
   748→  return `M-${String(max + 1).padStart(5, "0")}`;
   749→}

   751→function duplicateFieldErrors(db: DemoDb, phone: string | null | undefined, card: string | null | undefined, excludeId?: string) {
   752→  const errors: Record<string, string[]> = {};
   753→  if (phone && db.members.some((m) => m.phoneNumber === phone && m.id !== excludeId)) {
   754→    errors.phoneNumber = [`Phone number ${phone} is already registered to another member.`];
   755→  }
   756→  if (card && db.members.some((m) => m.cardCode === card && m.id !== excludeId)) {
   757→    errors.cardCode = [`Card ${card} is already assigned to another member.`];
   758→  }
   759→  return errors;
   760→}

   762→export async function demoSearchMembers(query: MemberQuery): Promise<PagedResult<Member>> {
   763→  await wait(180);
   764→  const db = loadDb();
   765→  const page = query.page && query.page > 0 ? query.page : 1;
   766→  const pageSize = query.pageSize && query.pageSize > 0 ? Math.min(query.pageSize, 100) : 20;
   767→
   768→  const term = query.search?.trim().toLowerCase() ?? "";
   769→  let items = [...db.members];
   770→
   771→  if (term) {
   772→    items = items.filter(
   773→      (m) =>
   774→        m.fullName.toLowerCase().includes(term) ||
   775→        m.code.toLowerCase().includes(term) ||
   776→        (m.phoneNumber ?? "").includes(term) ||
   777→        (m.cardCode ?? "").toLowerCase().includes(term),
   778→    );
   779→  }
   780→  if (query.status) {
   781→    items = items.filter((m) => m.status === query.status);
   782→  }
   783→  if (query.branchId) {
   784→    items = items.filter((m) => m.homeBranchId === query.branchId);
   785→  }
   786→
   787→  items.sort((a, b) => (a.joinDateUtc < b.joinDateUtc ? 1 : -1));
   788→  const totalCount = items.length;
   789→  const paged = items.slice((page - 1) * pageSize, page * pageSize);
   790→  return { items: paged, page, pageSize, totalCount };
   791→}

   800→export async function demoCreateMember(model: MemberWriteModel): Promise<Member> {
   801→  await wait(260);
   802→  const db = loadDb();
   803→
   804→  const errors: Record<string, string[]> = {};
   805→  if (!model.fullName.trim()) errors.fullName = ["Full name is required."];
   806→  if (!model.homeBranchId || !db.branches.some((b) => b.id === model.homeBranchId && b.isActive)) {
   807→    errors.homeBranchId = ["Unknown or inactive branch."];
   808→  }
   809→  Object.assign(errors, duplicateFieldErrors(db, model.phoneNumber?.trim() || null, model.cardCode?.trim() || null));
   810→  if (Object.keys(errors).length > 0) {
   811→    throw new GmsApiError(400, "validation.failed", "One or more validation errors occurred.", errors);
   812→  }
   813→
   814→  const member: Member = {
   815→    id: uid(),
   816→    code: nextMemberCode(db.members),
   817→    fullName: model.fullName.trim(),
   818→    phoneNumber: model.phoneNumber?.trim() || null,
   819→    gender: model.gender ?? "Unknown",
   820→    birthDateUtc: model.birthDate ? new Date(model.birthDate).toISOString() : null,
   821→    joinDateUtc: new Date().toISOString(),
   822→    homeBranchId: model.homeBranchId,
   823→    cardCode: model.cardCode?.trim() || null,
   824→    status: "Active",
   825→    notes: model.notes?.trim() || null,
   826→  };
   827→  db.members.push(member);
   828→  saveDb(db);
   829→  return member;
   830→}
